 INSERT INTO orders(customer_id, status, email, total, created_at, payload)
 SELECT
     1 + (random() * 9999)::int,
     (ARRAY['new', 'paid', 'cancelled'])[1 + (random() * 2)::int],
     'user' || g || '@example.com',
     round((10 + random() * 5000)::numeric, 2),
     now() - random() * interval '365 days',
     jsonb_build_object(
         'channel', (ARRAY['mobile', 'web', 'partner'])[1 + (random() * 2)::int],
         'country', (ARRAY['RU', 'KZ', 'BY', 'AM'])[1 + (random() * 3)::int]

 FROM generate_series(1, 200000) AS g;

 ANALYZE orders;

Что здесь показывает ACID:
Atomicity: если второе обновление завершится ошибкой, можно сделать ROLLBACK, и списание также исчезнет.
Consistency: CHECK (balance >= 0) не позволит получить отрицательный баланс.

Durability: после успешного COMMIT изменения должны пережить перезапуск сервера.

Ошибка внутри транзакции
 BEGIN;
 UPDATE accounts SET balance = balance - 5000 WHERE owner = 'Alice';
 -- ERROR: нарушено ограничение CHECK balance >= 0
 ROLLBACK;

После ошибки PostgreSQL переводит транзакцию в состояние aborted: до ROLLBACK обычные запросы выполнять
нельзя.

SAVEPOINT: частичный откат внутри транзакции
 BEGIN;
 UPDATE accounts SET balance = balance - 50 WHERE owner = 'Alice';

 SAVEPOINT before_bonus;
 UPDATE accounts SET balance = balance + 100 WHERE owner = 'Unknown';
 -- 0 строк обновлено: бизнес-логика может решить отменить только этот шаг
 ROLLBACK TO SAVEPOINT before_bonus;

 UPDATE accounts SET balance = balance + 50 WHERE owner = 'Bob';
 COMMIT;

SAVEPOINT не делает вложенную независимую транзакцию: окончательная устойчивость всё равно наступает
только после внешнего COMMIT.

2.2. Non-repeatable read: изменилась та же строка
Сессия A
 BEGIN ISOLATION LEVEL READ COMMITTED;
 SELECT balance FROM accounts WHERE owner = 'Alice'; -- допустим, 1000

 BEGIN;
 UPDATE accounts SET balance = 900 WHERE owner = 'Alice';
 COMMIT;

Сессия A
 SELECT balance FROM accounts WHERE owner = 'Alice'; -- уже 900
 COMMIT;

Одна и та же строка в одной транзакции прочитана дважды с разным результатом.

2.3. Phantom read: изменился состав множества
Сессия A
 BEGIN ISOLATION LEVEL READ COMMITTED;
 SELECT count(*) FROM orders WHERE status = 'new';

Сессия B
 INSERT INTO orders(customer_id, status, email, total)
 VALUES (42, 'new', 'phantom@example.com', 100);

Сессия A
 SELECT count(*) FROM orders WHERE status = 'new'; -- значение увеличилось
 COMMIT;

Изменилась не конкретная ранее прочитанная строка, а множество строк, удовлетворяющих условию.

2.4. Lost update: опасная схема “прочитал - посчитал - записал”
Плохой прикладной алгоритм:

 T2: прочитал balance = 1000
 T1: записал 900
 T2: записал 800

Итог 800, хотя ожидалось 700. Обновление T1 потеряно.

3.2. Read Uncommitted
Формально допускает грязные чтения — транзакция видит незакоммиченные изменения других.
В PostgreSQL этого уровня фактически нет: запрос READ UNCOMMITTED молча повышается до READ COMMITTED.
Причина — архитектурная: MVCC хранит версии строк, и механизма «показать чужую незакоммиченную версию»
просто не существует. Грязные чтения невозможны by design.

Эксперимент: PostgreSQL повышает уровень
 BEGIN ISOLATION LEVEL READ UNCOMMITTED;
 SHOW transaction_isolation;
 -- read uncommitted может отображаться как запрошенное имя,
 -- но видимость данных соответствует Read Committed.
 COMMIT;

Сессия A
 BEGIN;
 UPDATE accounts SET balance = 1 WHERE owner = 'Alice';
 -- COMMIT пока не выполняем

Сессия B
 BEGIN ISOLATION LEVEL READ UNCOMMITTED;
 SELECT balance FROM accounts WHERE owner = 'Alice';
 -- незакоммиченное значение 1 не видно
 COMMIT;

Retry на JDBC/Kotlin
 import java.sql.Connection
 import java.sql.SQLException

 fun <T> Connection.inSerializableTransaction(
     maxAttempts: Int = 4,
     block: (Connection) -> T
 ): T {
     val oldAutoCommit = autoCommit
     val oldIsolation = transactionIsolation

      try {
          repeat(maxAttempts) { attempt ->
              autoCommit = false
              transactionIsolation = Connection.TRANSACTION_SERIALIZABLE

              try {
                  val result = block(this)
                  commit()
                  return result
              } catch (e: SQLException) {
                  rollback()

                  val retryable = e.sqlState == "40001" || // serialization_failure
                                  e.sqlState == "40P01"    // deadlock_detected
                  if (!retryable || attempt == maxAttempts - 1) throw e
              }
          }
          error("unreachable")
      } finally {
          transactionIsolation = oldIsolation
          autoCommit = oldAutoCommit
      }
 }

4.2. Типы и режимы блокировок в PostgreSQL
По гранулярности: - Блокировки уровня строки (row-level) — берутся при UPDATE/DELETE/SELECT ... FOR
UPDATE. Не мешают чтению (читатели идут по MVCC-снимку), мешают конкурирующей записи той же строки. -
Блокировки уровня таблицы (table-level) — берутся операциями вроде ALTER TABLE, DROP, TRUNCATE, VACUUM
FULL, а также неявно обычными запросами (в слабых режимах).

Основные строковые режимы (от слабого к сильному): - FOR KEY SHARE — слабейший; защищает ключевые
столбцы от изменения/удаления (нужно для FK). - FOR SHARE — разделяемая; несколько транзакций могут
держать её одновременно, но пишущий будет ждать. - FOR NO KEY UPDATE — как UPDATE, но не трогает ключи. -
FOR UPDATE — эксклюзивная на строку; классика против lost update: «читаю для изменения — блокирую».
Совместимость (упрощённо): SHARE-режимы совместимы между собой, но не с эксклюзивными;
эксклюзивные не совместимы ни с чем. Именно матрица совместимости определяет, кто кого будет ждать.
Advisory locks (рекомендательные) — блокировки «по договорённости», не привязанные к строкам/таблицам,
а к числовому ключу: pg_advisory_lock(id). Полезны для прикладной сериализации (например, «одновременно
только один воркер обрабатывает пользователя N»). СУБД их не проверяет автоматически — соблюдать должно
приложение.

4.3. MVCC изнутри
Вместо блокировки читателей PostgreSQL хранит несколько версий строки (кортежей, tuples). У каждого
кортежа есть служебные поля: - xmin — id транзакции, которая создала эту версию. - xmax — id транзакции,
которая удалила/заменила эту версию (0, если версия живая).
Видимость. Транзакция при чтении сравнивает свой снимок (список активных на его момент транзакций) с
xmin/xmax кортежа и решает, видна ли ей эта версия. Грубо: версия видна, если создавшая транзакция уже
закоммичена до нашего снимка, а удаляющая — ещё нет (или откатилась). Отсюда: - Грязных чтений не
бывает — незакоммиченные версии просто не проходят проверку видимости. - Читатели не блокируют
писателей и наоборот — они работают с разными версиями.
UPDATE — это не изменение на месте. PostgreSQL помечает старую версию (проставляет xmax) и создаёт
новую версию строки. DELETE — только проставляет xmax. Значит, старые версии физически остаются в
таблице.
Мёртвые кортежи и VACUUM. Версии, которые уже не видны ни одной активной транзакции, — «мёртвые». Их
вычищает VACUUM / autovacuum, освобождая место под переиспользование и обновляя карту видимости. Без
вакуума таблица и индексы «распухают» (bloat), запросы замедляются. Также вакуум решает проблему
wraparound счётчика транзакций (xid 32-битный).
HOT-обновления (Heap-Only Tuples). Если UPDATE не меняет проиндексированные столбцы и новая версия
помещается на той же странице, PostgreSQL создаёт версию без обновления индексов (индекс продолжает

записи во всех обычных индексах.

 SELECT relname, n_tup_upd, n_tup_hot_upd

 WHERE relname = 'accounts';

 CREATE INDEX accounts_balance_idx ON accounts(balance);

Изменение balance уже обязано обновлять этот индекс, поэтому HOT для таких обновлений станет невозможен.

4.4. Дедлоки
Дедлок — транзакции ждут ресурсы по кругу:
 T1 держит A, хочет B
 T2 держит B, хочет A    →   цикл, обе ждут вечно

Детектирование. PostgreSQL не проверяет цикл при каждом ожидании (дорого). Транзакция, заблокированная
дольше deadlock_timeout (по умолчанию 1 с), запускает проверку графа ожиданий. Если найден цикл — одна
транзакция (жертва) откатывается:
 ERROR: deadlock detected

Приложение должно её повторить.

4.5. Пессимистичный vs оптимистичный подход
Пессимистичный: заранее беру блокировки, считая конфликт вероятным (FOR UPDATE, 2PL). Меньше откатов,
больше ожиданий/дедлоков. Хорош при высокой конкуренции за одни строки.
Оптимистичный: конфликт считаю редким, блокировок не беру, а на commit проверяю, не поменялось ли
(версионный столбец или Serializable/Repeatable Read с retry). Меньше ожиданий, но нужны повторы. Хорош при
низкой конкуренции.

5.2. B-дерево изнутри
По умолчанию CREATE INDEX создаёт B-tree (точнее B+-дерево).
Устройство на диске. Индекс — набор страниц (обычно по 8 КБ), организованных в дерево: - Корень и
внутренние узлы содержат разделяющие ключи и указатели на нижележащие страницы. Служат «дорожными
указателями». - Листья содержат сами ключи и указатели на строки (TID). В B+-дереве все данные — только в
листьях. - Дерево сбалансировано: путь от корня до любого листа одинаков. Глубина ≈ log(N), обычно 3–5
уровней даже на сотнях миллионов строк → максимум несколько чтений страниц на поиск. - Листья связаны в
двусвязный список → быстрый проход по диапазону и в обе стороны (для ORDER BY ASC/DESC).
Поиск WHERE x = 42: 1. Корень → бинарный поиск по отсортированным ключам страницы → выбираем ветвь. 2.
Спускаемся уровень за уровнем, каждый раз отбрасывая большую часть дерева. 3. В листе находим 42 и TID →
идём в таблицу. Это O(log n).
Вставка и расщепление страниц (page split). При вставке ключ идёт в нужный лист. Если лист переполнен —
он расщепляется на два, а разделяющий ключ поднимается в родителя (тот тоже может расщепиться, вплоть
до роста дерева на уровень). Отсюда: - fillfactor — насколько плотно заполнять страницы при создании (по
умолчанию 90% для B-tree). Меньше плотность → меньше расщеплений при будущих вставках, но больше
размер. - Частые вставки в случайные места → частые расщепления и фрагментация; вставки в конец
(возрастающий ключ, как serial/timestamp) → эффективно, расщепления только справа.
Почему LIKE 'foo%' использует индекс, а '%bar' — нет (главный пример на понимание): - Индекс
отсортирован по значению слева направо. Все строки на foo лежат подряд — это непрерывный диапазон
['foo', 'fop'). Дерево быстро находит начало и идёт по листьям. Работают col LIKE 'foo%', col ~ '^foo'. -
LIKE '%bar' ищет по окончанию — совпадения размазаны по всему дереву, диапазона нет, индекс бесполезен.

Но у возрастающего ключа может появиться “горячая” правая страница при очень высокой конкуренции, поэтому
нет универсального лучшего ключа для всех нагрузок.

5.3. Способы сканирования по индексу
Планировщик выбирает не только «индекс или seq scan», но и как по индексу идти:
  •    Index Scan — идём по индексу, для каждого совпадения прыгаем в heap за строкой. Хорош при малой
       выборке (высокая селективность).
  •    Index Only Scan — если все нужные столбцы есть в самом индексе, в heap идти не надо (проверяется
       ещё карта видимости — visibility map). Очень быстро. Достигается покрывающим индексом:
       CREATE INDEX idx ON t (a) INCLUDE (b, c);   -- b,c лежат в листьях, но не в ключе

  •    Bitmap Index Scan — сначала по индексу(ам) строим битовую карту нужных страниц heap, потом читаем
       heap по порядку страниц (последовательно, а не прыжками). Выгоден при средней выборке и при
       комбинации нескольких индексов (WHERE a=1 AND b=2 — два bitmap’а объединяются AND/OR).

5.6. Обслуживание индексов
Bloat — из-за MVCC индексы накапливают ссылки на мёртвые версии и распухают; лечится autovacuum, при
сильном распухании — REINDEX (можно REINDEX ... CONCURRENTLY, не блокируя запись).
Создание без блокировки: CREATE INDEX CONCURRENTLY строит индекс, не блокируя запись в таблицу (дольше,
но безопасно на проде).
Не индексируйте всё подряд: каждый индекс замедляет запись и мешает HOT-обновлениям. Индексируйте
под реальные запросы; лишние индексы удаляйте (их использование видно в pg_stat_user_indexes).
Проверка плана: EXPLAIN (ANALYZE, BUFFERS) <запрос> показывает, использовался ли индекс, каким сканом и
сколько строк реально прочитано. Основной инструмент диагностики.

Создание индекса на проде
 CREATE INDEX CONCURRENTLY orders_created_prod_idx
 ON orders(created_at);

Особенности:
команда дольше обычного CREATE INDEX;
не может выполняться внутри transaction block;
при сбое может остаться INVALID index, который нужно удалить или перестроить;
запись в таблицу не блокируется обычным длительным эксклюзивным lock, но краткие блокировки и нагрузка всё
равно есть.

5.7. Ключевые интуиции (чтобы не зубрить)
Порядок ключей в структуре определяет, что она ускорит. Отсортированное дерево (B-tree) → равенство,
диапазоны, префиксы, сортировка. Хеш → только равенство (порядок уничтожен). Инвертированный (GIN) →
«содержит элемент».
Индекс помогает, когда условие сужается до непрерывного участка структуры. Как только совпадения
«размазаны» (суффикс, только не-левый столбец составного индекса, функция без functional-индекса) — индекс
не применить.
Индекс ускоряет чтение, но замедляет запись и ест место. Ставить осознанно.
Малоселективные условия индекс не спасают — планировщик выберет seq scan.
MVCC ↔ индексы: обновление неиндексированных полей дешевле (HOT), лишние индексы этому мешают.


Разбор типичных ошибок
Ошибка: “Есть индекс, значит PostgreSQL обязан его использовать”.
Правильно: планировщик выбирает самый дешёвый план по оценке. Индекс может быть дороже Seq Scan.
Ошибка: “Индекс по (a,b) полностью бесполезен для b”.
Правильно: базово он хуже подходит без a; в PostgreSQL 18 возможен skip scan, но решение зависит от
статистики и стоимости.
Ошибка: “Index Only Scan никогда не читает таблицу”.
Правильно: при unset all-visible bit он делает heap fetch для проверки MVCC-видимости.
Ошибка: “VACUUM уменьшает файл таблицы”.
Правильно: обычный VACUUM освобождает место внутри файла для повторного использования. Возврат места
ОС обычно требует переписывания (VACUUM FULL, CLUSTER или другие методы).
Ошибка: “Serializable означает отсутствие ошибок”.
Правильно: он обеспечивает корректный результат ценой штатных откатов и обязательного retry.

Что чаще всего спрашивают
Различие dirty / non-repeatable / phantom + lost update / write skew.

Задание 6. Исправить lost update
Плохой код:
 SELECT balance
 newBalance = balance - amount
 UPDATE accounts SET balance = newBalance

Решение A: атомарный UPDATE
 UPDATE accounts
 SET balance = balance - :amount
 WHERE id = :id AND balance >= :amount
 RETURNING balance;

Решение B: optimistic locking
 UPDATE accounts
 SET balance = :new_balance,
     version = version + 1
 WHERE id = :id AND version = :old_version;

Решение C: pessimistic locking
 SELECT balance FROM accounts WHERE id = :id FOR UPDATE;

Выбор зависит от частоты конфликтов и длительности операции.

Блокировки и MVCC
11. Блокирует ли обычный SELECT UPDATE?
Обычно нет: SELECT читает подходящую MVCC-версию. Исключения связаны с явными locks и тяжёлыми DDL
locks.
12. Что блокирует SELECT FOR UPDATE?
Конкурирующие изменения/удаления и несовместимые row-lock режимы для выбранных строк.
13. Когда использовать SKIP LOCKED?
Для очередей и параллельных воркеров, где допустимо временно пропустить занятую задачу.
14. Что такое xmin и xmax?
Системные поля tuple: транзакция создания и транзакция удаления/замены/блокировки версии. Реальная
семантика xmax может включать MultiXact.
15. Почему UPDATE создаёт новую версию?
MVCC позволяет старым снимкам продолжать видеть старую версию, не блокируя читателей.
16. Зачем VACUUM?
Убирать недоступные версии для повторного использования места, обслуживать visibility map и предотвращать
XID wraparound.
17. Что такое HOT?
Оптимизация UPDATE без новых записей в обычных индексах, когда не меняются индексированные столбцы и
новая версия помещается на той же странице.
18. Как избежать дедлока?
Единый порядок захвата ресурсов, короткие транзакции, нужный lock сразу и retry при 40P01.
